cmake_minimum_required(VERSION 3.22)
project(Azazel6 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(AZAZEL6_FRAMEWORK "Build pinned PSPRecomp tooling and runtime" OFF)
set(AZAZEL6_FRAMEWORK_SOURCE "${CMAKE_SOURCE_DIR}/local/deps/yakumo" CACHE PATH "Pinned Yakumo checkout")
if(ANDROID AND NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Azazel6 currently targets Android arm64-v8a only")
endif()
add_library(azazel6_math STATIC profiles/tekken6/host/aspect.cpp)
target_include_directories(azazel6_math PUBLIC profiles/tekken6/include)
if(MSVC)
    target_compile_options(azazel6_math PRIVATE /W4 /WX)
else()
    target_compile_options(azazel6_math PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    add_executable(azazel6_math_tests tests/aspect_test.cpp)
    target_link_libraries(azazel6_math_tests PRIVATE azazel6_math)
    add_test(NAME azazel6_math COMMAND azazel6_math_tests)
endif()
if(AZAZEL6_FRAMEWORK)
    find_package(Git REQUIRED)
    file(READ "${CMAKE_SOURCE_DIR}/dependencies.json" dependency_json)
    string(JSON expected_revision GET "${dependency_json}" yakumo revision)
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${AZAZEL6_FRAMEWORK_SOURCE}" rev-parse HEAD
        OUTPUT_VARIABLE actual_revision OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE git_result)
    if(NOT git_result EQUAL 0 OR NOT actual_revision STREQUAL expected_revision)
        message(FATAL_ERROR "Framework revision mismatch; run python tools/framework.py fetch")
    endif()
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${AZAZEL6_FRAMEWORK_SOURCE}" status --porcelain --untracked-files=no
        OUTPUT_VARIABLE framework_changes OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE status_result)
    if(NOT status_result EQUAL 0 OR NOT framework_changes STREQUAL "")
        message(FATAL_ERROR "Framework checkout has tracked modifications")
    endif()
    # Never select Yakumo's game profile. Azazel6 owns its profile independently.
    set(PSPRECOMP_PROFILE "" CACHE STRING "Framework-only upstream build" FORCE)
    set(PSPRECOMP_BUILD_TESTS ${BUILD_TESTING} CACHE BOOL "Framework tests" FORCE)
    set(PSPRECOMP_MSVC_MP_JOBS 1 CACHE STRING "Bound compiler parallelism")
    add_subdirectory("${AZAZEL6_FRAMEWORK_SOURCE}" "framework")
    add_subdirectory(profiles/tekken6)
endif()
