cmake_minimum_required(VERSION 3.22)
project(Azazel6PortableKitResearch LANGUAGES C CXX)
get_filename_component(AZAZEL6_ROOT "../../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(AZAZEL6_PORTABLEKIT_SOURCE "${AZAZEL6_ROOT}/local/deps/portablekit" CACHE PATH "Unmodified pinned PortableKit checkout")
find_package(Git REQUIRED)
file(READ "${AZAZEL6_ROOT}/dependencies.json" lock)
string(JSON expected GET "${lock}" portablekit revision)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${AZAZEL6_PORTABLEKIT_SOURCE}" rev-parse HEAD
    OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE result TIMEOUT 15)
if(NOT result EQUAL 0 OR NOT actual STREQUAL expected)
    message(FATAL_ERROR "PortableKit revision mismatch; refusing to modify the checkout")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${AZAZEL6_PORTABLEKIT_SOURCE}" status --porcelain
    OUTPUT_VARIABLE dirty RESULT_VARIABLE result TIMEOUT 15)
if(NOT result EQUAL 0 OR NOT dirty STREQUAL "")
    message(FATAL_ERROR "PortableKit checkout must be clean")
endif()
# No implicit downloads, renderer, installer or media dependency in this probe.
set(PORTABLEKIT_RENDERER OFF CACHE BOOL "Headless research" FORCE)
set(PORTABLEKIT_FFMPEG OFF CACHE STRING "No media downloads" FORCE)
set(PSPRECOMP_MSVC_MP_JOBS 1 CACHE STRING "Bound compiler concurrency")
add_subdirectory("${AZAZEL6_PORTABLEKIT_SOURCE}" portablekit)
portablekit_add_game(azazel6_portablekit_probe PROFILE_DIR "${CMAKE_CURRENT_SOURCE_DIR}"
    HOST_MAIN_NAME unused_portablekit_main SOURCES probe.cpp)
target_sources(azazel6_portablekit_probe PRIVATE "${AZAZEL6_ROOT}/profiles/tekken6/host/bootstrap.cpp")
target_include_directories(azazel6_portablekit_probe PRIVATE "${AZAZEL6_ROOT}/profiles/tekken6/include")
enable_testing()
add_executable(azazel6_portablekit_bootstrap_tests "${AZAZEL6_ROOT}/profiles/tekken6/tests/bootstrap_test.cpp"
    "${AZAZEL6_ROOT}/profiles/tekken6/host/bootstrap.cpp")
target_include_directories(azazel6_portablekit_bootstrap_tests PRIVATE "${AZAZEL6_ROOT}/profiles/tekken6/include")
target_link_libraries(azazel6_portablekit_bootstrap_tests PRIVATE psprecomp_core)
add_test(NAME portablekit_bootstrap_compatibility COMMAND azazel6_portablekit_bootstrap_tests)
add_executable(azazel6_portablekit_clock_tests "${AZAZEL6_ROOT}/profiles/tekken6/tests/clock_conversion_test.cpp")
target_include_directories(azazel6_portablekit_clock_tests PRIVATE "${AZAZEL6_ROOT}/profiles/tekken6/include")
target_link_libraries(azazel6_portablekit_clock_tests PRIVATE psprecomp_core)
add_test(NAME portablekit_clock_compatibility COMMAND azazel6_portablekit_clock_tests)
