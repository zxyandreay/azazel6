cmake_minimum_required(VERSION 3.22)
project(Azazel6PortableKitResearch LANGUAGES C CXX)
get_filename_component(AZAZEL6_ROOT "../../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(AZAZEL6_PORTABLEKIT_SOURCE "${AZAZEL6_ROOT}/local/deps/portablekit" CACHE PATH "Unmodified pinned PortableKit checkout")
find_package(Git REQUIRED)
file(READ "${AZAZEL6_ROOT}/dependencies.json" lock)
string(JSON expected GET "${lock}" portablekit revision)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${AZAZEL6_PORTABLEKIT_SOURCE}" rev-parse HEAD
    OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE result TIMEOUT 15)
if(NOT result EQUAL 0 OR NOT actual STREQUAL expected)
    message(FATAL_ERROR "PortableKit revision mismatch; refusing to modify the checkout")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${AZAZEL6_PORTABLEKIT_SOURCE}" status --porcelain
    OUTPUT_VARIABLE dirty RESULT_VARIABLE result TIMEOUT 15)
if(NOT result EQUAL 0 OR NOT dirty STREQUAL "")
    message(FATAL_ERROR "PortableKit checkout must be clean")
endif()
# Renderer evaluation is explicit; normal compatibility runs remain headless.
option(AZAZEL6_PORTABLEKIT_RENDERER "Build the optional renderer contract probe" OFF)
set(PORTABLEKIT_RENDERER ${AZAZEL6_PORTABLEKIT_RENDERER} CACHE BOOL "Explicit renderer research" FORCE)
set(AZAZEL6_PORTABLEKIT_FFMPEG OFF CACHE STRING "Optional real ATRAC/movie decoder: OFF, system, or bundled")
set_property(CACHE AZAZEL6_PORTABLEKIT_FFMPEG PROPERTY STRINGS OFF system bundled)
if(NOT AZAZEL6_PORTABLEKIT_FFMPEG MATCHES "^(OFF|system|bundled)$")
    message(FATAL_ERROR "AZAZEL6_PORTABLEKIT_FFMPEG must be OFF, system, or bundled")
endif()
set(PORTABLEKIT_FFMPEG ${AZAZEL6_PORTABLEKIT_FFMPEG} CACHE STRING "Explicit media research" FORCE)
set(PSPRECOMP_MSVC_MP_JOBS 1 CACHE STRING "Bound compiler concurrency")
add_subdirectory("${AZAZEL6_PORTABLEKIT_SOURCE}" portablekit)
set(AZAZEL6_PORTABLEKIT_AOT_SOURCE "" CACHE FILEPATH "Locally generated exact-EBOOT entry corpus")
set(aot_sources)
set(aot_options)
if(AZAZEL6_PORTABLEKIT_AOT_SOURCE)
    get_filename_component(aot_source "${AZAZEL6_PORTABLEKIT_AOT_SOURCE}" ABSOLUTE)
    get_filename_component(local_root "${AZAZEL6_ROOT}/local" REALPATH)
    cmake_path(IS_PREFIX local_root "${aot_source}" NORMALIZE source_is_local)
    if(NOT source_is_local OR NOT EXISTS "${aot_source}")
        message(FATAL_ERROR "AOT source must exist under this repository's ignored local/ directory")
    endif()
    get_filename_component(aot_dir "${aot_source}" DIRECTORY)
    if(NOT EXISTS "${aot_dir}/report.json")
        message(FATAL_ERROR "AOT source lacks its exact-input generation report")
    endif()
    file(READ "${aot_dir}/report.json" aot_report)
    file(READ "${AZAZEL6_ROOT}/profiles/tekken6/config/profile.json" game_profile)
    string(JSON aot_revision GET "${aot_report}" portablekit_revision)
    string(JSON aot_executable GET "${aot_report}" executable_sha256)
    string(JSON aot_entry GET "${aot_report}" entry)
    string(JSON aot_source_sha GET "${aot_report}" generated_source_sha256)
    string(JSON expected_executable GET "${game_profile}" executable sha256)
    file(SHA256 "${aot_source}" actual_source_sha)
    if(NOT aot_revision STREQUAL expected OR NOT aot_executable STREQUAL expected_executable OR
       NOT aot_entry STREQUAL "0x08804124" OR NOT aot_source_sha STREQUAL actual_source_sha)
        message(FATAL_ERROR "AOT source/report does not match the pinned Tekken entry corpus")
    endif()
    list(APPEND aot_sources "${aot_source}")
    list(APPEND aot_options NO_CORPUS)
    if(MSVC)
        set_source_files_properties("${aot_source}" PROPERTIES COMPILE_OPTIONS "/O2;/Ob0;/bigobj")
    else()
        set_source_files_properties("${aot_source}" PROPERTIES COMPILE_OPTIONS "-O2;-g0")
    endif()
endif()
portablekit_add_game(azazel6_portablekit_probe ${aot_options} PROFILE_DIR "${CMAKE_CURRENT_SOURCE_DIR}"
    HOST_MAIN_NAME unused_portablekit_main SOURCES probe.cpp ge_contracts.cpp renderer_contracts.cpp ${aot_sources})
if(aot_sources)
    target_compile_definitions(azazel6_portablekit_probe PRIVATE AZAZEL6_ENTRY_AOT=1)
endif()
if(AZAZEL6_PORTABLEKIT_RENDERER)
    get_target_property(renderer_definitions azazel6_portablekit_probe COMPILE_DEFINITIONS)
    if(NOT "PORTABLEKIT_HAS_RENDERER=1" IN_LIST renderer_definitions)
        message(FATAL_ERROR "Explicit renderer evaluation requires SDL3, Vulkan and glslang; no headless fallback")
    endif()
endif()
target_sources(azazel6_portablekit_probe PRIVATE
    "${AZAZEL6_ROOT}/profiles/tekken6/host/bootstrap.cpp"
    "${AZAZEL6_ROOT}/profiles/tekken6/host/media.cpp"
    "${AZAZEL6_ROOT}/profiles/tekken6/host/media_hash.cpp")
target_include_directories(azazel6_portablekit_probe PRIVATE "${AZAZEL6_ROOT}/profiles/tekken6/include")
enable_testing()
add_test(NAME portablekit_ge_compatibility COMMAND azazel6_portablekit_probe --test-ge)

    "${AZAZEL6_ROOT}/profiles/tekken6/host/bootstrap.cpp")
target_include_directories(azazel6_portablekit_bootstrap_tests PRIVATE "${AZAZEL6_ROOT}/profiles/tekken6/include")
target_link_libraries(azazel6_portablekit_bootstrap_tests PRIVATE psprecomp_core)
add_test(NAME portablekit_bootstrap_compatibility COMMAND azazel6_portablekit_bootstrap_tests)
add_executable(azazel6_portablekit_clock_tests "${AZAZEL6_ROOT}/profiles/tekken6/tests/clock_conversion_test.cpp")
target_include_directories(azazel6_portablekit_clock_tests PRIVATE "${AZAZEL6_ROOT}/profiles/tekken6/include")
target_link_libraries(azazel6_portablekit_clock_tests PRIVATE psprecomp_core)
add_test(NAME portablekit_clock_compatibility COMMAND azazel6_portablekit_clock_tests)
add_executable(azazel6_portablekit_wakeup_tests cancel_wakeup_test.cpp)
target_include_directories(azazel6_portablekit_wakeup_tests PRIVATE
    "${AZAZEL6_PORTABLEKIT_SOURCE}/host" "${AZAZEL6_PORTABLEKIT_SOURCE}/include")
target_link_libraries(azazel6_portablekit_wakeup_tests PRIVATE psprecomp_core)
add_test(NAME portablekit_wakeup_compatibility COMMAND azazel6_portablekit_wakeup_tests)
