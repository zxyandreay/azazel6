set(AZAZEL6_ENTRY_DIR "" CACHE PATH "Directory produced by tools/entry_probe.py")
option(AZAZEL6_PROBE_FAST_BUILD "Disable entry-probe optimization for coverage research" OFF)
add_library(azazel6_bootstrap STATIC host/bootstrap.cpp host/threads.cpp host/memory.cpp host/module.cpp)
target_include_directories(azazel6_bootstrap PUBLIC include)
target_link_libraries(azazel6_bootstrap PUBLIC psprecomp_core)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    add_executable(azazel6_bootstrap_tests tests/bootstrap_test.cpp)
    target_link_libraries(azazel6_bootstrap_tests PRIVATE azazel6_bootstrap)
    add_test(NAME azazel6_bootstrap COMMAND azazel6_bootstrap_tests)
    add_executable(azazel6_thread_tests tests/threads_test.cpp)
    target_link_libraries(azazel6_thread_tests PRIVATE azazel6_bootstrap)
    add_test(NAME azazel6_threads COMMAND azazel6_thread_tests)
    add_executable(azazel6_memory_tests tests/memory_test.cpp)
    target_link_libraries(azazel6_memory_tests PRIVATE azazel6_bootstrap)
    add_test(NAME azazel6_memory COMMAND azazel6_memory_tests)
    add_executable(azazel6_module_tests tests/module_test.cpp)
    target_link_libraries(azazel6_module_tests PRIVATE azazel6_bootstrap)
    add_test(NAME azazel6_module COMMAND azazel6_module_tests)
    add_executable(azazel6_control_flow_tests tests/control_flow_test.cpp)
    target_link_libraries(azazel6_control_flow_tests PRIVATE azazel6_bootstrap)
    add_test(NAME azazel6_control_flow COMMAND azazel6_control_flow_tests)
endif()
if(AZAZEL6_ENTRY_DIR)
    if(NOT EXISTS "${AZAZEL6_ENTRY_DIR}/entry.cpp" OR NOT EXISTS "${AZAZEL6_ENTRY_DIR}/probe_config.hpp" OR NOT EXISTS "${AZAZEL6_ENTRY_DIR}/imports.cpp")
        message(FATAL_ERROR "Generate the local entry probe first")
    endif()
    add_executable(azazel6_entry_probe host/entry_probe.cpp "${AZAZEL6_ENTRY_DIR}/entry.cpp" "${AZAZEL6_ENTRY_DIR}/imports.cpp")
    target_include_directories(azazel6_entry_probe PRIVATE "${AZAZEL6_ENTRY_DIR}")
    target_link_libraries(azazel6_entry_probe PRIVATE azazel6_bootstrap)
    if(AZAZEL6_PROBE_FAST_BUILD)
        if(MSVC)
            target_compile_options(azazel6_entry_probe PRIVATE /Od)
        else()
            target_compile_options(azazel6_entry_probe PRIVATE -O0)
        endif()
    endif()
endif()
