cmake_minimum_required(VERSION 3.22)
project(Azazel6HostProbe LANGUAGES C CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "The host probe requires Android ARM64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(AZAZEL6_SDL_SOURCE "${repo}/local/deps/sdl" CACHE PATH "Pinned SDL checkout")
find_package(Git REQUIRED)
file(READ "${repo}/dependencies.json" lock)
string(JSON expected GET "${lock}" sdl revision)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${AZAZEL6_SDL_SOURCE}" rev-parse HEAD
    OUTPUT_VARIABLE revision OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE result)
if(NOT result EQUAL 0 OR NOT revision STREQUAL expected)
    message(FATAL_ERROR "SDL revision mismatch")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${AZAZEL6_SDL_SOURCE}" status --porcelain
    OUTPUT_VARIABLE changes OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE result)
if(NOT result EQUAL 0 OR NOT changes STREQUAL "")
    message(FATAL_ERROR "SDL checkout has modifications")
endif()
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${AZAZEL6_SDL_SOURCE}" sdl)
add_library(main SHARED main.cpp presenter.cpp)
target_include_directories(main PRIVATE "${repo}/profiles/tekken6/include")
target_link_libraries(main PRIVATE SDL3::SDL3 vulkan android log)
target_compile_options(main PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-missing-field-initializers)
target_link_options(main PRIVATE "-Wl,-z,max-page-size=16384")
